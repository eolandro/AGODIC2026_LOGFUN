# Definimos la función llamada fun y recibe un número N
def fun(N):
    # Si N es igual a 0, llegamos al caso base
    if N == 0:
        # La función indica que f(0) = 1
        return 1
    # Si N no es 0, hacemos la llamada recursiva
    else:
        # Calculamos el valor anterior, es decir, f(N-1)
        anterior = fun(N - 1)
        # Aplicamos la fórmula: f(N) = f(N-1)^N - 2*f(N-1)
        return anterior ** N - 2 * anterior


# Llamamos a la función con 0 y mostramos el resultado
print(fun(0))
# Llamamos a la función con 1 y mostramos el resultado
print(fun(1))
# Llamamos a la función con 2 y mostramos el resultado
print(fun(2))
# Llamamos a la función con 3 y mostramos el resultado
print(fun(3))

# Creamos el grafo con sus conexiones
Grafo = [
    ["AZQ","BXQ"],
    ["BXQ","CYQ"],
    ["CYQ","DWQ"],
    ["DWQ","EVQ"],
    ["EVQ","FUQ"],
    ["FUQ","GRX"],
    ["GRX","HSX"],
    ["HSX","ITQ"],
    ["ITQ","JUQ"],
    ["JUQ","AZQ"]
]

# Esta función busca qué elemento está antes del elemento actual
def buscaAnterior(grafo, inicio, posicion):
    # Revisamos si el segundo elemento de la posición actual
    # es igual al elemento que estamos buscando
    if grafo[posicion][1] == inicio:
        # Si coincide, regresamos el elemento que estaba antes
        return grafo[posicion][0]
    # Si todavía no encontramos el elemento
    else:
        # Avanzamos a la siguiente posición del grafo
        # y volvemos a llamar a la función
        return buscaAnterior(grafo, inicio, posicion + 1)

# Esta función imprime el recorrido del grafo en orden inverso
def impGrafCirR(grafo, inicio, limite):
    # Si llegamos al elemento límite, imprimimos ese elemento
    if inicio == limite:
        # Mostramos el elemento límite
        print(inicio)
    # Si todavía no llegamos al límite
    else:
        # Buscamos cuál es el elemento anterior al actual
        siguiente = buscaAnterior(grafo, inicio, 0)
        # Llamamos nuevamente a la función para seguir recorriendo
        # el grafo hasta llegar al límite
        impGrafCirR(grafo, siguiente, limite)
        # Cuando la recursividad regresa, imprimimos el elemento actual
        print(inicio)

# Iniciamos el recorrido desde AZQ hasta llegar a EVQ
impGrafCirR(Grafo, "AZQ", "EVQ")
# Dejamos una línea en blanco para separar los resultados

def composicion_recursiva(Valor, ListaFunciones):
    if not ListaFunciones:
        return Valor
    match ListaFunciones:
        case [PF]:
            return PF(Valor)
        case [PF, *FF]:
            return composicion_recursiva(PF(Valor), FF)
        case _:
            return Valor

def fun1(R):
    return R.split(",")

def fun2(L):
    match L:
        case [P, *F]:
            return [int(P)] + F
        case _:
            return []

def contar(L, Valor):
    if not L:
        return 0

    match L:
        case [P, *F]:
            if int(P) == Valor:
                return 1 + contar(F, Valor)

            return contar(F, Valor)

        case _:
            return 0

def fun3(L):
    match L:
        case [ID, Nombre, *F]:
            return [
                ID,
                Nombre,
                contar(F, 0),
                contar(F, 1)
            ]

        case _:
            return []

R = "29590095,ROMAY TACITURNO RENE FERNANDA,0,1,1,1,1,1,1,0"

# 3

print(fun(3))
# 192900153618
"""
def funZ(N):
    match N:
        case int():
            if N < 0:
                return None
            if N == 0:
                return 3
            else:
                return funZ(N - 1) ** 3 - 3 * funZ(N - 1)
        case _:
            return None

print(funZ(0))

def impMat(matriz):
